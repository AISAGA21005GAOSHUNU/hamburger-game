// =============================== // 🔧 遊戲設定 // =============================== const MAX_SPINS = 8; // 每局最多轉幾次 const SPIN_DURATION_MS = 3000; // 食材設定（可自由擴充） const SEGMENTS = [ { label: "牛肉排", icon: "🥩", type: "meat", weight: 3 }, { label: "雞肉排", icon: "🍗", type: "meat", weight: 3 }, { label: "豬肉排", icon: "🍖", type: "meat", weight: 2 }, { label: "素肉排", icon: "🫘", type: "veggie", weight: 1 }, { label: "起司", icon: "🧀", type: "cheese", weight: 4 }, { label: "雙倍起司", icon: "🧀", type: "cheese", weight: 2 }, { label: "生菜", icon: "🥬", type: "veggie", weight: 4 }, { label: "番茄", icon: "🍅", type: "veggie", weight: 4 }, { label: "酸黃瓜", icon: "🥒", type: "veggie", weight: 3 }, { label: "培根", icon: "🥓", type: "meat", weight: 2 }, { label: "醬汁", icon: "🥫", type: "sauce", weight: 3 }, { label: "超大漢堡", icon: "🍔", type: "special", weight: 1 } ]; // =============================== // DOM // =============================== const wheel = document.getElementById("wheel"); const wheelLabels = document.getElementById("wheelLabels"); const spinButton = document.getElementById("spinButton"); const resetButton = document.getElementById("resetBurgerButton"); const remainingText = document.getElementById("remainingSpinsText"); const currentResult = document.getElementById("currentResult"); const historyList = document.getElementById("historyList"); const burgerLayers = document.getElementById("burgerLayers"); const burgerStatus = document.getElementById("burgerStatus"); // ✨ 選擇彈窗 const modal = document.getElementById("choiceModal"); const modalTitle = document.getElementById("modalTitle"); const modalDesc = document.getElementById("modalDescription"); const btnAdd = document.getElementById("btnAdd"); const btnSkip = document.getElementById("btnSkip"); // =============================== // 狀態 // =============================== let currentRotation = 0; let isSpinning = false; let remainingSpins = MAX_SPINS; let burger = []; let pendingSegment = null; let lastActiveIndex = null; // =============================== // 初始化 // =============================== initWheel(); updateRemaining(); // =============================== // 建立轉盤背景（色塊） // =============================== function initWheel() { const total = SEGMENTS.length; const step = 360 / total; // 背景色塊 wheel.style.background = `conic-gradient( ${SEGMENTS.map((s, i) => { const a1 = i * step; const a2 = (i + 1) * step; const color = getColorByType(s.type); return `${color} ${a1}deg ${a2}deg`; }).join(",")} )`; // icon 在圓周上 wheelLabels.innerHTML = ""; SEGMENTS.forEach((s, i) => { const angle = i * step + step / 2; const el = document.createElement("div"); el.className = "segment-label"; el.dataset.index = i; el.innerHTML = `${s.icon}`; el.style.transform = `rotate(${angle}deg) translate(0, -78%) rotate(${-angle}deg)`; wheelLabels.appendChild(el); }); } // 顏色對應 function getColorByType(type) { switch (type) { case "meat": return "#ffb74d"; case "veggie": return "#a5d6a7"; case "cheese": return "#ffe082"; case "sauce": return "#ffab91"; case "special": return "#ce93d8"; default: return "#ffe0b2"; } } // =============================== // 抽獎 → 權重隨機 // =============================== function getRandomIndex() { const total = SEGMENTS.reduce((sum, s) => sum + s.weight, 0); let r = Math.random() * total; for (let i = 0; i < SEGMENTS.length; i++) { r -= SEGMENTS[i].weight; if (r <= 0) return i; } return SEGMENTS.length - 1; } // =============================== // 開始轉盤 // =============================== spinButton.addEventListener("click", () => { if (isSpinning) return; if (remainingSpins <= 0) return; isSpinning = true; spinButton.disabled = true; const index = getRandomIndex(); const step = 360 / SEGMENTS.length; const fullSpins = 3 + Math.floor(Math.random() * 3); const target = fullSpins * 360 - (index * step + step / 2); currentRotation = target; wheel.style.transform = `rotate(${target}deg)`; // 清除前一個高亮 if (lastActiveIndex !== null) { wheelLabels.querySelector(`[data-index="${lastActiveIndex}"]`)?.classList.remove("active"); } setTimeout(() => { isSpinning = false; spinButton.disabled = false; remainingSpins--; updateRemaining(); pendingSegment = SEGMENTS[index]; lastActiveIndex = index; showChoiceModal(pendingSegment); }, SPIN_DURATION_MS); }); function updateRemaining() { remainingText.textContent = remainingSpins; } // =============================== // ✨ 彈窗邏輯：加入 or 隱藏 // =============================== function showChoiceModal(seg) { modalTitle.textContent = `抽到：${seg.label}`; modalDesc.innerHTML = `
${seg.icon}
是否加入漢堡？`; modal.classList.remove("hidden"); // 高亮 wheelLabels.querySelector(`[data-index="${lastActiveIndex}"]`).classList.add("active"); } function closeModal() { modal.classList.add("hidden"); } // 玩家選擇加入 btnAdd.addEventListener("click", () => { if (!pendingSegment) return; addIngredient(pendingSegment); addHistory(pendingSegment); currentResult.innerHTML = `加入：${pendingSegment.icon} ${pendingSegment.label}`; pendingSegment = null; closeModal(); }); // 玩家選擇略過 btnSkip.addEventListener("click", () => { currentResult.innerHTML = `略過：${pendingSegment.icon} ${pendingSegment.label}`; pendingSegment = null; closeModal(); }); // =============================== // 加入漢堡 // =============================== function addIngredient(seg) { burger.push(seg); const el = document.createElement("div"); el.className = "burger-layer ingredient-new"; el.textContent = `${seg.icon} ${seg.label}`; burgerLayers.appendChild(el); burgerStatus.textContent = `目前共有 ${burger.length} 層食材。`; } // 歷史紀錄 function addHistory(seg) { const item = document.createElement("div"); item.className = "history-item"; item.textContent = `${seg.icon} ${seg.label}（${new Date().toLocaleTimeString()}）`; historyList.prepend(item); } // =============================== // 重置整局 // =============================== resetButton.addEventListener("click", resetGame); function resetGame() { remainingSpins = MAX_SPINS; updateRemaining(); burger = []; burgerLayers.innerHTML = ""; historyList.innerHTML = ""; currentResult.textContent = "尚未開始。"; burgerStatus.textContent = "尚未加入任何食材。"; wheel.style.transform = "rotate(0deg)"; }
